1.Acceptance Rate By Date

SELECT
    s.date,
    ROUND(
        COUNT(a.user_id_sender) * 1.0 / COUNT(s.user_id_sender),
        2
    ) AS acceptance_rate
FROM fb_friend_requests s
LEFT JOIN fb_friend_requests a
ON s.user_id_sender = a.user_id_sender
AND s.user_id_receiver = a.user_id_receiver
AND a.action = 'accepted'
WHERE s.action = 'sent'
GROUP BY s.date
HAVING COUNT(a.user_id_sender) > 0
ORDER BY s.date;

2.Daily Revenue
WITH purchases AS (
    SELECT
        transaction_id,
        transaction_date,
        amount
    FROM product_sales
    WHERE product_id = 'PROD-2891'
      AND country = 'US'
      AND status = 'completed'
      AND type = 'purchase'
      AND transaction_date BETWEEN '2025-04-15' AND '2025-04-28'
),

refunds AS (
    SELECT
        p.transaction_date,
        r.amount
    FROM product_sales r
    JOIN purchases p
      ON r.original_transaction_id = p.transaction_id
    WHERE r.type = 'refund'
      AND r.status = 'completed'
),

daily_values AS (

    SELECT
        transaction_date,
        amount AS revenue
    FROM purchases

    UNION ALL

    SELECT
        transaction_date,
        -amount AS revenue
    FROM refunds
),

dates AS (
    SELECT generate_series(
        DATE '2025-04-15',
        DATE '2025-04-28',
        INTERVAL '1 day'
    )::date AS transaction_date
)

SELECT
    d.transaction_date,
    COALESCE(SUM(v.revenue), 0) AS daily_net_revenue
FROM dates d
LEFT JOIN daily_values v
ON d.transaction_date = v.transaction_date
GROUP BY d.transaction_date
ORDER BY d.transaction_date;


3.Finding User Purchases
WITH first_purchase AS (
    SELECT
        user_id,
        MIN(created_at) AS first_purchase_date
    FROM amazon_transactions
    GROUP BY user_id
)

SELECT DISTINCT a.user_id
FROM amazon_transactions a
JOIN first_purchase f
ON a.user_id = f.user_id
WHERE a.created_at > f.first_purchase_date
AND a.created_at <= f.first_purchase_date + INTERVAL '7 days'
ORDER BY a.user_id;



4.Search Click Success Rate by User Segment
WITH latest_date AS (
    SELECT MAX(event_timestamp::date) AS max_date
    FROM search_events
),

user_segments AS (
    SELECT
        a.user_id,
        CASE
            WHEN a.registration_date >= (
                SELECT max_date - INTERVAL '30 days'
                FROM latest_date
            )
            THEN 'new'
            ELSE 'existing'
        END AS user_segment
    FROM accounts a
),

searches AS (
    SELECT
        se.event_id,
        se.user_id,
        se.session_id,
        se.event_timestamp
    FROM search_events se
    WHERE se.event_type = 'search'
),

successful_searches AS (
    SELECT DISTINCT s.event_id
    FROM searches s
    JOIN search_events c
      ON s.session_id = c.session_id
     AND s.user_id = c.user_id
     AND c.event_type = 'click'
     AND c.event_timestamp > s.event_timestamp
     AND c.event_timestamp <= s.event_timestamp + INTERVAL '30 seconds'
)

SELECT
    us.user_segment,
    COUNT(s.event_id) AS total_searches,
    COUNT(ss.event_id) AS successful_searches,
    ROUND(
        COUNT(ss.event_id)::numeric
        / COUNT(s.event_id),
        2
    ) AS success_rate
FROM searches s
JOIN user_segments us
  ON s.user_id = us.user_id
LEFT JOIN successful_searches ss
  ON s.event_id = ss.event_id
GROUP BY us.user_segment
ORDER BY us.user_segment;
